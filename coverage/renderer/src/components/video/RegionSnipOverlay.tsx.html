function resizeRect(orig: Rect, handle: Handle, p: Pt): Rect {
  let { x, y, w, h } = orig
  const right = x + w
  const bottom = y + h
  if (handle.includes('w')) { x = Math.min(p.x, right - MIN_NATIVE); w = right - x }

  if (handle.includes('s')) { h = Math.max(MIN_NATIVE, p.y - y) }
  return { x, y, w, h }
